Weekly Fitness Progress Tracker · Tech Spec

Stack ที่แนะนำ: Next.js (App Router) + PostgreSQL ผ่าน Prisma, เก็บรูปบน S3/Cloud Storage. หลักการสำคัญคือเก็บข้อมูลดิบระดับเซ็ต แล้วคำนวณสรุปรายสัปดาห์ด้วย SQL (หรือ materialized view) เพื่อให้กราฟดึงข้อมูลได้ในคำขอเดียว

1. Data Schema (PostgreSQL)

users / goals
ข้อมูลกายภาพ + เป้าหมายหลักและตัวเลข
exercises / routines
คลังท่าฝึก และ Quick Template
workouts → workout_sets / cardio_logs
บันทึกรายวัน ระดับเซ็ต
body_metrics / progress_photos
Check-in สัปดาห์ละครั้ง
CREATE TYPE goal_type AS ENUM ('lose_weight','build_muscle','endurance');
CREATE TYPE sex_type  AS ENUM ('male','female','other');

CREATE TABLE users (
  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  email        text UNIQUE NOT NULL,
  height_cm    numeric(5,1),
  birth_date   date,
  sex          sex_type,
  week_start   smallint DEFAULT 1,          -- 1 = Monday
  created_at   timestamptz DEFAULT now()
);

CREATE TABLE goals (
  id                uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id           uuid REFERENCES users(id) ON DELETE CASCADE,
  type              goal_type NOT NULL,
  target_weight_kg  numeric(5,1),
  workouts_per_week smallint DEFAULT 4,
  start_weight_kg   numeric(5,1),
  active            boolean DEFAULT true,
  created_at        timestamptz DEFAULT now()
);

CREATE TABLE exercises (
  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id      uuid REFERENCES users(id),   -- NULL = global library
  name         text NOT NULL,
  muscle_group text,
  increment_kg numeric(4,2) DEFAULT 2.5      -- ใช้คำนวณ progressive overload
);

CREATE TABLE routines (
  id      uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid REFERENCES users(id) ON DELETE CASCADE,
  name    text NOT NULL                       -- 'Chest Day', 'Leg Day'
);
CREATE TABLE routine_exercises (
  routine_id   uuid REFERENCES routines(id) ON DELETE CASCADE,
  exercise_id  uuid REFERENCES exercises(id),
  position     smallint,
  default_sets smallint DEFAULT 3,
  default_reps smallint DEFAULT 8,
  PRIMARY KEY (routine_id, exercise_id)
);

CREATE TABLE workouts (
  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id      uuid REFERENCES users(id) ON DELETE CASCADE,
  routine_id   uuid REFERENCES routines(id),
  kind         text CHECK (kind IN ('strength','cardio')),
  performed_at timestamptz NOT NULL DEFAULT now(),
  duration_min smallint,
  calories     integer
);

CREATE TABLE workout_sets (
  id          uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  workout_id  uuid REFERENCES workouts(id) ON DELETE CASCADE,
  exercise_id uuid REFERENCES exercises(id),
  set_no      smallint NOT NULL,
  weight_kg   numeric(6,2) NOT NULL,
  reps        smallint NOT NULL,
  rpe         smallint CHECK (rpe BETWEEN 1 AND 10),
  note        text
);
CREATE INDEX ON workout_sets (exercise_id);

CREATE TABLE cardio_logs (
  workout_id   uuid PRIMARY KEY REFERENCES workouts(id) ON DELETE CASCADE,
  activity     text,                          -- run / bike / walk
  distance_km  numeric(6,2),
  duration_min numeric(6,1),
  calories     integer
);

CREATE TABLE body_metrics (
  id         uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id    uuid REFERENCES users(id) ON DELETE CASCADE,
  week_start date NOT NULL,                   -- 1 แถวต่อสัปดาห์
  weight_kg  numeric(5,1),
  chest_cm   numeric(5,1), waist_cm numeric(5,1), hip_cm numeric(5,1),
  arm_cm     numeric(5,1), thigh_cm numeric(5,1),
  UNIQUE (user_id, week_start)
);

CREATE TABLE progress_photos (
  id         uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id    uuid REFERENCES users(id) ON DELETE CASCADE,
  week_start date NOT NULL,
  pose       text DEFAULT 'front',            -- front / side / back
  storage_key text NOT NULL,                  -- S3 object key
  UNIQUE (user_id, week_start, pose)
);

Auto-populate (Progressive Overload)

// GET /api/exercises/:id/previous
export async function GET(req, { params }) {
  const userId = await requireUser(req);
  const [last] = await sql`
    SELECT ws.weight_kg, ws.reps, ws.rpe, w.performed_at,
           COUNT(*) OVER (PARTITION BY w.id) AS sets
    FROM workout_sets ws
    JOIN workouts w ON w.id = ws.workout_id
    WHERE w.user_id = ${userId}
      AND ws.exercise_id = ${params.id}
      AND w.performed_at < date_trunc('week', now())   -- สัปดาห์ก่อนหน้าลงไป
    ORDER BY w.performed_at DESC, ws.weight_kg DESC
    LIMIT 1`;
  if (!last) return Response.json(null);

  const { increment_kg } = await getExercise(params.id);
  // RPE ≤ 8 → เพิ่มน้ำหนัก, RPE ≥ 9 → คงน้ำหนัก เพิ่ม 1 rep
  const suggestion = last.rpe >= 9
    ? { weight_kg: last.weight_kg, reps: last.reps + 1 }
    : { weight_kg: +last.weight_kg + +increment_kg, reps: last.reps };

  return Response.json({ previous: last, suggestion });
}

Weekly summary + strength chart

-- /api/stats/weekly-summary
SELECT date_trunc('week', w.performed_at)::date      AS week_start,
       COUNT(DISTINCT w.id)                          AS total_workouts,
       COALESCE(SUM(ws.weight_kg * ws.reps), 0)      AS total_volume_kg,
       SUM(DISTINCT w.duration_min)                  AS total_minutes,
       SUM(DISTINCT w.calories)                      AS total_calories
FROM workouts w
LEFT JOIN workout_sets ws ON ws.workout_id = w.id
WHERE w.user_id = $1
  AND w.performed_at >= date_trunc('week', now()) - ($2 - 1) * interval '1 week'
GROUP BY 1 ORDER BY 1;
-- หมายเหตุ: production ควร aggregate duration/calories แยก subquery
-- เพื่อไม่ให้ SUM(DISTINCT) ตัดค่าซ้ำ

-- /api/stats/strength  (top set + Δ vs สัปดาห์ก่อน)
WITH weekly AS (
  SELECT date_trunc('week', w.performed_at)::date AS week_start,
         MAX(ws.weight_kg)                         AS top_kg,
         MAX(ws.weight_kg * (1 + ws.reps / 30.0))  AS est_1rm   -- Epley
  FROM workout_sets ws JOIN workouts w ON w.id = ws.workout_id
  WHERE w.user_id = $1 AND ws.exercise_id = $2
  GROUP BY 1
)
SELECT *, top_kg - LAG(top_kg) OVER (ORDER BY week_start) AS delta_kg
FROM weekly ORDER BY week_start DESC LIMIT $3;

4. UX notes สำหรับใช้งานในยิม

• ปุ่มและช่องกรอกทั้งหมดสูง ≥ 44px, ใช้ inputmode="decimal" เพื่อเปิด numpad ทันที
• เซ็ตถูก prefill ด้วยค่าแนะนำ ผู้ใช้แค่กด ✓ ถ้าทำได้ตามเป้า ลดการพิมพ์ระหว่างพักเซ็ต
• บันทึกแบบ offline-first (IndexedDB) แล้ว sync เมื่อมีสัญญาณ เพราะยิมหลายที่สัญญาณไม่ดี
• ช่องสัดส่วนร่างกายใช้ค่าสัปดาห์ก่อนเป็น placeholder ถ้าเว้นว่างจะคงค่าเดิม
